Program to Check Even or Odd

#include <stdio.h>

int main() {
    int number;

    printf("Enter a number: ");
    scanf("%d", &number);

    if (number % 2 == 0) {
        printf("The number is even.");
    } else {
        printf("The number is odd.");
    }

    return 0;
}

This program takes two numbers from the user, adds them, stores the result in the sum variable, and displays the result.

Sample PPS Lab Program

Program to Add Two Numbers

#include <stdio.h>

int main() {
    int a, b, sum;

    printf("Enter two numbers: ");
    scanf("%d %d", &a, &b);

    sum = a + b;

    printf("Sum = %d", sum);

    return 0;
}
